\begin{proposition}\label{prop:odd-product}
Let $a$ and $b$ be odd integers. Then $ab$ is odd.
\end{proposition}
\begin{proof}
Write $a=2r+1$ and $b=2s+1$ with integers $r,s$.
Then $ab=2(2rs+r+s)+1$.
\end{proof}
For instance, $3\cdot5=15$.
